CMAKE_MINIMUM_REQUIRED(VERSION 3.10)
PROJECT(RTS)

SET(CMAKE_EXPORT_COMPILE_COMMANDS ON)

IF(APPLE)
    LIST(APPEND CMAKE_PREFIX_PATH "/opt/homebrew")
ENDIF()

FIND_PACKAGE(raylib CONFIG REQUIRED)

ADD_EXECUTABLE(main "./main.c")
SET_PROPERTY(TARGET main PROPERTY C_STANDARD 23)

ADD_LIBRARY(grid_utils STATIC "./src/grid_utils.c")
SET_PROPERTY(TARGET grid_utils PROPERTY C_STANDARD 23)

TARGET_INCLUDE_DIRECTORIES(main PRIVATE "./include")
TARGET_INCLUDE_DIRECTORIES(grid_utils PRIVATE "./include")

TARGET_LINK_LIBRARIES(grid_utils PUBLIC raylib)
TARGET_LINK_LIBRARIES(main PRIVATE grid_utils)
